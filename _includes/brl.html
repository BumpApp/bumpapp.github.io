{%- comment -%}Formats a number as Brazilian decimal with two places: 20.0 -> 20,00. Usage: {% include brl.html v=price %}{%- endcomment -%} {%- assign c = include.v | times: 100.0 | round -%} {%- assign w = c | divided_by: 100 -%} {%- assign f = c | modulo: 100 | prepend: "0" | slice: -2, 2 -%} {{ w }},{{ f }}
