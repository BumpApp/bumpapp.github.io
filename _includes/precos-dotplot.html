{%- comment -%} Dot plot of price per GB for every pack in a prices data file, small and monthly packs as two groups, with Bump's ceiling and Wi-Fi rate as reference lines. Params: lang ("pt", "en" or "es"); data (data file name, default "precos_br"); cur ("brl" default, or "cop"); xmax and step (axis max and tick step in the currency per GB, defaults 28 and 4). Rows are sorted by price per GB via zero-padded string keys, since Liquid can't sort on a computed value. The key string ends in a comma; Liquid's split (Ruby String#split) drops trailing empty elements, so no empty key results. {%- endcomment -%} {%- assign dname = include.data | default: "precos_br" -%}{%- assign d = site.data[dname] -%} {%- assign cur = include.cur | default: "brl" -%} {%- if cur == "cop" -%}{%- assign fmt = "cop.html" -%}{%- assign sym = "$" -%}{%- assign rate = d.bump.credits_per_cop -%}{%- else -%}{%- assign fmt = "brl.html" -%}{%- assign sym = "R$" -%}{%- assign rate = d.bump.credits_per_brl -%}{%- endif -%} {%- assign lang = include.lang | default: "en" -%} {%- case lang -%} {%- when "pt" -%}{%- assign dec = "," -%}{%- assign t_small = "Pacotes pequenos" -%}{%- assign t_large = "Pacotes mensais" -%}{%- assign t_max = "máx." -%} {%- when "es" -%}{%- assign dec = "," -%}{%- assign t_small = "Paquetes pequeños" -%}{%- assign t_large = "Paquetes mensuales" -%}{%- assign t_max = "máx." -%} {%- else -%}{%- assign dec = "." -%}{%- assign t_small = "Small packs" -%}{%- assign t_large = "Monthly packs" -%}{%- assign t_max = "max" -%} {%- endcase -%} {%- assign bump_gb = d.bump.credits_per_mb | times: 1024.0 | divided_by: rate -%} {%- assign wifi_gb = d.bump.credits_per_mb_wifi | times: 1024.0 | divided_by: rate -%} {%- assign small = "" -%}{%- assign large = "" -%} {%- for p in d.packs -%} {%- assign k = p.price | times: 102400.0 | divided_by: p.mb | round | prepend: "0000000000" | slice: -10, 10 | append: "|" | append: forloop.index0 | append: "," -%} {%- if p.size == "small" -%}{%- assign small = small | append: k -%}{%- else -%}{%- assign large = large | append: k -%}{%- endif -%} {%- endfor -%} {%- assign small = small | split: "," | sort | reverse -%} {%- assign large = large | split: "," | sort | reverse -%} {%- assign n = small.size | plus: large.size -%} {%- assign W = 840 -%}{%- assign L = 252 -%}{%- assign R = 60 -%}{%- assign rowH = 22 -%} {%- assign xmax = include.xmax | default: 28 -%}{%- assign step = include.step | default: 4 -%} {%- assign ticks = xmax | divided_by: step -%} {%- assign plotW = W | minus: L | minus: R -%} {%- assign top = 30 -%}{%- assign gapG = 34 -%} {%- assign H = n | times: rowH | plus: top | plus: gapG | plus: gapG | plus: 30 -%} {%- assign plotTop = top | minus: 8 -%}{%- assign plotBot = H | minus: 26 -%} {%- assign xb = bump_gb | times: plotW | divided_by: xmax | plus: L -%} {%- assign xw = wifi_gb | times: plotW | divided_by: xmax | plus: L -%}
